# syntax=docker/dockerfile:1
# Layered session images. The default target is base; rust and mobile are
# additive targets selected by the project image mapping.

ARG UBUNTU_VERSION=24.04
ARG NODE_VERSION=22.14.0
ARG PNPM_VERSION=10.12.4
ARG UV_VERSION=0.7.12
ARG GRAFT_VERSION=0.1.0
ARG CLAUDE_CODE_VERSION=2.1.289
ARG CODEX_VERSION=0.160.0
ARG OPENCODE_VERSION=1.18.31
ARG RUST_VERSION=1.94.0
ARG RUST_TARGETS="x86_64-unknown-linux-gnu,aarch64-unknown-linux-gnu"
ARG FLUTTER_VERSION=3.35.3
ARG ANDROID_CMDLINE_TOOLS_VERSION=13114758
ARG ANDROID_PLATFORM_TOOLS_VERSION=35.0.2
ARG CHROMIUM_VERSION=1:134.0.6998.88-1~deb12u1

FROM rust:1.94-slim AS session-agent-build
WORKDIR /src
COPY Cargo.toml Cargo.lock ./
COPY contract/Cargo.toml contract/Cargo.toml
COPY core/Cargo.toml core/Cargo.toml
COPY server/Cargo.toml server/Cargo.toml
COPY session-agent/Cargo.toml session-agent/Cargo.toml
COPY session-agent session-agent
RUN cargo build --locked --release -p vogt-session-agent

FROM ubuntu:${UBUNTU_VERSION} AS base
ARG NODE_VERSION
ARG PNPM_VERSION
ARG UV_VERSION
ARG GRAFT_VERSION
ARG CLAUDE_CODE_VERSION
ARG CODEX_VERSION
ARG OPENCODE_VERSION
ARG TARGETARCH

LABEL org.opencontainers.image.title="vogt-session-base" \
      org.opencontainers.image.description="Base image for Vogt coding sessions" \
      org.opencontainers.image.source="https://github.com/TheDancingDeveloper-org/vogt"

ENV DEBIAN_FRONTEND=noninteractive \
    LANG=C.UTF-8 \
    HOME=/home/vogt \
    UV_TOOL_BIN_DIR=/usr/local/bin \
    NPM_CONFIG_PREFIX=/usr/local \
    PATH=/usr/local/bin:/usr/local/sbin:/home/vogt/.local/bin:/opt/rust/cargo/bin:/opt/flutter/bin:/opt/android-sdk/platform-tools:/opt/android-sdk/cmdline-tools/latest/bin:$PATH

# Keep the base deterministic at the tool level. Package indexes are not
# retained in the layer, and no credentials or runtime configuration is copied.
RUN apt-get update \
    && apt-get install -y --no-install-recommends \
       ca-certificates curl git git-lfs jq ripgrep fd-find build-essential \
       pkg-config libssl-dev python3 python3-pip python3-venv python3-dev \
       openssh-client sudo unzip zip xz-utils zstd \
    && find /var/lib/apt/lists -type f -delete

# Node is installed from the official, versioned tarball rather than a moving
# distribution channel. pnpm and uv are exact tool pins.
RUN arch="$(dpkg --print-architecture)" \
    && case "$arch" in amd64) node_arch=x64 ;; arm64) node_arch=arm64 ;; *) echo "unsupported architecture: $arch" >&2; exit 1 ;; esac \
    && curl -fsSL "https://nodejs.org/dist/v${NODE_VERSION}/node-v${NODE_VERSION}-linux-${node_arch}.tar.xz" -o /tmp/node.tar.xz \
    && mkdir -p /opt/node \
    && tar -xJf /tmp/node.tar.xz --strip-components=1 -C /opt/node \
    && ln -s /opt/node/bin/node /usr/local/bin/node \
    && ln -s /opt/node/bin/npm /usr/local/bin/npm \
    && ln -s /opt/node/bin/npx /usr/local/bin/npx \
    && npm install --global "pnpm@${PNPM_VERSION}" \
    && curl -fsSL "https://github.com/astral-sh/uv/releases/download/${UV_VERSION}/uv-x86_64-unknown-linux-gnu.tar.gz" -o /tmp/uv.tar.gz \
    && tar -xzf /tmp/uv.tar.gz -C /tmp \
    && install -m 0755 /tmp/uv-x86_64-unknown-linux-gnu/uv /usr/local/bin/uv \
    && install -m 0755 /tmp/uv-x86_64-unknown-linux-gnu/uvx /usr/local/bin/uvx \
    && find /tmp -maxdepth 1 -type f \( -name 'node.tar.xz' -o -name 'uv.tar.gz' \) -delete

# Graft is pinned by release, not fetched from a mutable branch. The image
# remains usable without a graft token; authentication is supplied at runtime.
RUN curl -fsSL "https://github.com/TheDancingDeveloper-org/graft/releases/download/v${GRAFT_VERSION}/graft-x86_64-unknown-linux-musl.tar.gz" -o /tmp/graft.tar.gz \
    && tar -xzf /tmp/graft.tar.gz -C /usr/local/bin --strip-components=1 \
    && chmod 0755 /usr/local/bin/graft \
    && ln -s /usr/local/bin/graft /usr/local/bin/g \
    && find /tmp -maxdepth 1 -type f -name 'graft.tar.gz' -delete

# Agent CLIs are pinned in the image, while their credentials are always
# brokered after startup. opencode is installed from npm to avoid a curl|sh.
RUN npm install --global \
      "@anthropic-ai/claude-code@${CLAUDE_CODE_VERSION}" \
      "@openai/codex@${CODEX_VERSION}" \
      "opencode-ai@${OPENCODE_VERSION}"

RUN if ! getent group vogt >/dev/null; then groupadd vogt; fi \
    && if ! id -u vogt >/dev/null 2>&1; then useradd --gid vogt --create-home --shell /bin/bash vogt; fi \
    && install -d -o vogt -g vogt /home/vogt/Working /home/vogt/.config /home/vogt/.cache

# These are intentionally small runtime hooks. WI-3163 supplies the real
# session entrypoint; this placeholder makes the image safe to smoke-test.
COPY --chmod=0755 deploy/agent-auth.sh /usr/local/bin/vogt-agent-auth
COPY --chmod=0755 deploy/claude-session-hook.sh /usr/local/bin/vogt-claude-session-hook
COPY --chmod=0755 deploy/policy-hook.sh /usr/local/bin/vogt-policy-hook
COPY --chmod=0755 deploy/session-entrypoint.sh /usr/local/bin/vogt-session-entrypoint
COPY --from=session-agent-build /src/target/release/vogt-session-agent /usr/local/bin/vogt-session-agent

USER vogt
WORKDIR /home/vogt/Working
ENTRYPOINT ["/usr/local/bin/vogt-session-entrypoint"]

FROM base AS rust
ARG RUST_VERSION
ARG RUST_TARGETS
ARG TARGETARCH
LABEL org.opencontainers.image.title="vogt-session-rust"
ENV RUSTUP_HOME=/opt/rust/rustup \
    CARGO_HOME=/opt/rust/cargo \
    SCCACHE_DIR=/var/cache/sccache
USER root
RUN install -d -o vogt -g vogt /opt/rust /var/cache/sccache \
    && su - vogt -c "curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh -s -- -y --default-toolchain ${RUST_VERSION} --profile minimal" \
    && su - vogt -c "rustup target add $(printf '%s' '${RUST_TARGETS}' | tr ',' ' ')" \
    && apt-get update \
    && apt-get install -y --no-install-recommends clang lld cmake protobuf-compiler \
    && find /var/lib/apt/lists -type f -delete \
    && su - vogt -c "cargo install --locked cargo-nextest --version 0.9.102" \
    && su - vogt -c "rustup component add rust-analyzer"
# This config is a placeholder for a runtime-provided sccache client. No cache
# credentials or cache contents are baked into the image.
RUN printf '%s\n' '[build]' '# rustc-wrapper = "/usr/local/bin/sccache"' > /etc/cargo-config.toml
USER vogt

FROM rust AS mobile
ARG FLUTTER_VERSION
ARG ANDROID_CMDLINE_TOOLS_VERSION
ARG ANDROID_PLATFORM_TOOLS_VERSION
ARG CHROMIUM_VERSION
LABEL org.opencontainers.image.title="vogt-session-mobile"
USER root
ENV ANDROID_HOME=/opt/android-sdk ANDROID_SDK_ROOT=/opt/android-sdk FLUTTER_ROOT=/opt/flutter
RUN apt-get update \
    && apt-get install -y --no-install-recommends chromium=${CHROMIUM_VERSION} xwayland libgl1-mesa-dri \
    && find /var/lib/apt/lists -type f -delete \
    && curl -fsSL "https://storage.googleapis.com/flutter_infra_release/releases/stable/linux/flutter_linux_${FLUTTER_VERSION}-stable.tar.xz" -o /tmp/flutter.tar.xz \
    && tar -xJf /tmp/flutter.tar.xz -C /opt \
    && chown -R vogt:vogt /opt/flutter \
    && install -d -o vogt -g vogt /opt/android-sdk/cmdline-tools \
    && curl -fsSL "https://dl.google.com/android/repository/commandlinetools-linux-${ANDROID_CMDLINE_TOOLS_VERSION}_latest.zip" -o /tmp/android.zip \
    && unzip -q /tmp/android.zip -d /opt/android-sdk/cmdline-tools \
    && mv /opt/android-sdk/cmdline-tools/cmdline-tools /opt/android-sdk/cmdline-tools/latest \
    && yes | /opt/android-sdk/cmdline-tools/latest/bin/sdkmanager --licenses >/dev/null \
    && /opt/android-sdk/cmdline-tools/latest/bin/sdkmanager "platform-tools" "platforms;android-35" "build-tools;35.0.0" \
    && chown -R vogt:vogt /opt/android-sdk \
    && find /tmp -maxdepth 1 -type f \( -name 'flutter.tar.xz' -o -name 'android.zip' \) -delete
USER vogt
RUN flutter config --no-analytics && flutter precache --android
