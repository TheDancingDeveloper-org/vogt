# The session user is uid/gid 1000. Ubuntu images may already ship a user at
# 1000, which is removed so vogt owns the id. /run/vogt (the session token) and
# /work (the clone) are created here because the non-root user cannot create
# them under / at runtime. The base may already have a root-owned /home/vogt,
# which useradd leaves alone, so the home is chowned explicitly.
RUN if existing="$(getent passwd 1000 | cut -d: -f1)" && [ -n "$existing" ]; then userdel --remove "$existing"; fi \
    && if existing="$(getent group 1000 | cut -d: -f1)" && [ -n "$existing" ]; then groupdel "$existing"; fi \
    && groupadd --gid 1000 vogt \
    && useradd --uid 1000 --gid 1000 --create-home --shell /bin/bash vogt \
    && cp -rn /etc/skel/. /home/vogt/ \
    && chown -R 1000:1000 /home/vogt \
    && install -d -o 1000 -g 1000 -m 0700 /run/vogt /work \
    && install -d -o 1000 -g 1000 /home/vogt/Working /home/vogt/.config /cache/xdg /cache/npm /cache/pnpm /cache/pip /cache/uv

# The Codex managed policy is only a policy if the session user can read it:
# `requirements.toml` being 0644 says nothing when its parent /etc/codex is
# root-only, and the session agent then refuses to start Codex at all
# ("cannot be read (Permission denied)"). The base image may already carry an
# /etc/codex with such a mode, so set both and prove the read as vogt — the
# same rule the merged image applies (engine/Dockerfile, WI-1143).
RUN mkdir -p /etc/codex \
    && chown root:root /etc/codex \
    && chmod 0755 /etc/codex \
    && chown root:root /etc/codex/requirements.toml \
    && chmod 0644 /etc/codex/requirements.toml \
    && su - vogt -c 'test -r /etc/codex/requirements.toml'

# RUSTUP_AUTO_INSTALL=0: the session pod's root filesystem is read-only, so a
# rust-toolchain.toml that names another channel or version must fail with
# rustup's own error rather than try (and fail) to install it at run time.
# A rust-toolchain.toml with `channel = "stable"` must also resolve here, so the
# pinned toolchain must answer to the name `stable` too. rustup refuses
# `toolchain link stable`, so the RUN below adds a symlink of the toolchain
# directory under that name. Both names then point at the pinned ${RUST_VERSION}
# and nothing is downloaded at run time.
ENV RUSTUP_HOME=/opt/rust/rustup \
    CARGO_HOME=/cache/cargo \
    SCCACHE_DIR=/var/cache/sccache \
    RUSTUP_AUTO_INSTALL=0
USER root
RUN install -d -o vogt -g vogt /opt/rust /opt/rust/cargo /opt/rust/rustup /cache/cargo /var/cache/sccache \
    && case "$TARGETARCH" in amd64) rustup_arch=x86_64; rustup_sha="$RUSTUP_X64_SHA256" ;; arm64) rustup_arch=aarch64; rustup_sha="$RUSTUP_ARM64_SHA256" ;; *) echo "unsupported architecture: $TARGETARCH" >&2; exit 1 ;; esac \
    && curl --proto '=https' --tlsv1.2 -fsSL "https://static.rust-lang.org/rustup/archive/${RUSTUP_VERSION}/${rustup_arch}-unknown-linux-gnu/rustup-init" -o /tmp/rustup-init \
    && echo "${rustup_sha}  /tmp/rustup-init" | sha256sum -c - \
    && chown vogt:vogt /tmp/rustup-init \
    && chmod 0755 /tmp/rustup-init \
    && su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo /tmp/rustup-init -y --default-toolchain ${RUST_VERSION} --profile default" \
    && find /tmp -maxdepth 1 -type f -name rustup-init -delete \
    && su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo rustup component add rust-src" \
    && ln -s "${RUST_VERSION}-${rustup_arch}-unknown-linux-gnu" "/opt/rust/rustup/toolchains/stable-${rustup_arch}-unknown-linux-gnu" \
    && su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo rustup target add $(printf '%s' "${RUST_TARGETS}" | tr ',' ' ')" \
    && for v in $(printf '%s' "${RUST_EXTRA_VERSIONS}" | tr ',' ' '); do su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo rustup toolchain install $v --profile default --component rust-src --target $(printf '%s' "${RUST_TARGETS}")" || exit 1; done \
    && apt-get update \
    && apt-get install -y --no-install-recommends clang lld cmake protobuf-compiler \
    && find /var/lib/apt/lists -type f -delete \
    && su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo cargo install --locked cargo-nextest --version 0.9.102" \
    && su - vogt -c "RUSTUP_HOME=/opt/rust/rustup CARGO_HOME=/opt/rust/cargo rustup component add rust-analyzer" \
    && case "${TARGETARCH}" in \
         amd64) \
           curl -fsSL -o /tmp/kache.tgz "https://github.com/kunobi-ninja/kache/releases/download/${KACHE_VERSION}/kache-x86_64-unknown-linux-musl.tar.gz" \
             && echo "${KACHE_SHA256}  /tmp/kache.tgz" | sha256sum -c - \
             && install -d /tmp/kache \
             && tar -xzf /tmp/kache.tgz -C /tmp/kache \
             && install -m 0755 /tmp/kache/kache /tmp/kache/kache-engine /usr/local/bin/ \
             && find /tmp/kache.tgz /tmp/kache -delete \
             && kache --version ;; \
         *) echo "kache ${KACHE_VERSION} has no release for ${TARGETARCH}" >&2; exit 1 ;; \
       esac
# This config is a placeholder for a runtime-provided sccache client. No cache
# credentials or cache contents are baked into the image.
RUN printf '%s\n' '[build]' '# rustc-wrapper = "/usr/local/bin/sccache"' > /etc/cargo-config.toml
USER 1000:1000

    && chown -R vogt:vogt /opt/android-sdk \
    && find /tmp -maxdepth 1 -type f \( -name 'flutter.tar.xz' -o -name 'android.zip' \) -delete
