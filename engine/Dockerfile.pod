# The session user is uid/gid 1000. Ubuntu images may already ship a user at
# 1000, which is removed so vogt owns the id. /run/vogt (the session token) and
# /work (the clone) are created here because the non-root user cannot create
# them under / at runtime. The base may already have a root-owned /home/vogt,
# which useradd leaves alone, so the home is chowned explicitly.
RUN if existing="$(getent passwd 1000 | cut -d: -f1)" && [ -n "$existing" ]; then userdel --remove "$existing"; fi \
    && if existing="$(getent group 1000 | cut -d: -f1)" && [ -n "$existing" ]; then groupdel "$existing"; fi \
    && groupadd --gid 1000 vogt \
    && useradd --uid 1000 --gid 1000 --create-home --shell /bin/bash vogt \
    && cp -rn /etc/skel/. /home/vogt/ \
    && chown -R 1000:1000 /home/vogt \
    && install -d -o 1000 -g 1000 -m 0700 /run/vogt /work \
    && install -d -o 1000 -g 1000 /home/vogt/Working /home/vogt/.config /cache/xdg /cache/npm /cache/pnpm /cache/pip /cache/uv

# The Codex managed policy is only a policy if the session user can read it:
# `requirements.toml` being 0644 says nothing when its parent /etc/codex is
# root-only, and the session agent then refuses to start Codex at all
# ("cannot be read (Permission denied)"). The base image may already carry an
# /etc/codex with such a mode, so set both and prove the read as vogt — the
# same rule the merged image applies (engine/Dockerfile, WI-1143).
RUN mkdir -p /etc/codex \
    && chown root:root /etc/codex \
    && chmod 0755 /etc/codex \
    && chown root:root /etc/codex/requirements.toml \
    && chmod 0644 /etc/codex/requirements.toml \
    && su - vogt -c 'test -r /etc/codex/requirements.toml'

# The driven-session permission policy (WI-926, WI-3243): the session agent
# starts Claude Code with --settings pointing here, as the merged image's
# engine does for a host session. Root-owned and read-only to the session user.
# No --chmod: BuildKit applies it to the directories this creates too, and a

ENTRYPOINT ["/usr/local/bin/vogt-session-entrypoint"]
